definition{
  name: memberString,
  parameters: [value : String, values : (List String)],
  result: Bool,
  body: (match values
    | List.nil => false
    | List.cons head tail => (if (prim equal value head : Bool) then true else (memberString value tail))),
  axioms: [],
  recursive_argument: values
}

definition{
  name: allMembers,
  parameters: [values : (List String), ids : (List String)],
  result: Bool,
  body: (match values
    | List.nil => true
    | List.cons head tail => (if (memberString head ids) then (allMembers tail ids) else false)),
  axioms: [],
  recursive_argument: values
}

definition{
  name: distinctStrings,
  parameters: [values : (List String)],
  result: Bool,
  body: (match values
    | List.nil => true
    | List.cons head tail => (if (memberString head tail) then false else (distinctStrings tail))),
  axioms: [],
  recursive_argument: values
}

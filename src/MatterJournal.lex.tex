 (if ((prim equal operation "withdraw" : Bool) && (count =? 6)) then
 (MatterSession::withdraw session (numberField fields 1) (roleField (fieldAt fields 3)) (fieldAt fields 2) (reference fields 4)) else
 (if (((prim equal operation "ingest" : Bool) || (prim equal operation "ingest-csv" : Bool)) && (count =? 6)) then
 (if (prim equal operation "ingest-csv" : Bool) then (MatterSession::ingestCsv session (numberField fields 1) (roleField (fieldAt fields 3)) (fieldAt fields 2) (fieldAt fields 4) (Foundation.Utf8::encode (fieldAt fields 5))) else (MatterSession::ingest session (numberField fields 1) (roleField (fieldAt fields 3)) (fieldAt fields 2) (fieldAt fields 4) (Foundation.Utf8::encode (fieldAt fields 5)))) else
 (if ((prim equal operation "assert" : Bool) && (count =? 6)) then
 (MatterSession::recordAssertion session (numberField fields 1) (roleField (fieldAt fields 3)) (fieldAt fields 2) (fieldAt fields 4) (fieldAt fields 5)) else
 (if ((prim equal operation "source" : Bool) && (count =? 7)) then
 (MatterSession::reviseSource session (numberField fields 1) (roleField (fieldAt fields 3)) (fieldAt fields 2) (reference fields 4) (Foundation.Utf8::encode (fieldAt fields 6))) else
 (if ((prim equal operation "review" : Bool) && ((count =? 9) && ((prim equal (fieldAt fields 8) "true" : Bool) || (prim equal (fieldAt fields 8) "false" : Bool)))) then

definition{

 body: (let empty : MatterSession::Session := (MatterSession::initial "");
 (if (8388608 <? (Foundation.Bytes::byteLength (Foundation.Utf8::encode journal))) then (MatterSession::refuse empty "journal-limit") else
 (match (decodeJournalFields journal 514) | Option.none => (MatterSession::refuse empty "journal-encoding") | Option.some fields =>
 (match (project fields values) | List.nil => (MatterSession::refuse empty "journal-version") | List.cons version body =>
 (if (not (prim equal version "prism-legal/journal/1" : Bool)) then (MatterSession::refuse empty "journal-version") else
 (match body | List.nil => (MatterSession::refuse empty "journal-title") | List.cons title commands =>
 (if (not (MatterSession::boundedText title 256)) then (MatterSession::refuse empty "journal-title") else
 (let result : MatterSession::Outcome := (replayCommands commands (MatterSession::initial title));
 (if (prim equal (project result code) "" : Bool) then result else (MatterSession::refuse empty (project result code))))))))))),
 executable: true, axioms: [Classical.choice, Quot.sound, propext]
}

 (if (not ((project (project (project original session) matter) revision) <? 512)) then (record JournalResult {code := "journal-count", journal := journal, session := (project original session)}) else
 (let updated : MatterSession::Outcome := (applyCommand (project original session) command);
 (if (not (prim equal (project updated code) "" : Bool)) then (record JournalResult {code := (project updated code), journal := journal, session := (project original session)}) else
 (let next : String := (MatterText::concat journal (MatterInference::frame command));
 (if (8388608 <? (Foundation.Bytes::byteLength (Foundation.Utf8::encode next))) then (record JournalResult {code := "journal-limit", journal := journal, session := (project original session)}) else
 (record JournalResult {code := "", journal := next, session := (project updated session)})))))))),

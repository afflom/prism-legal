definition{
 name: foldLoop, parameters: [fuel : Nat, input : Bytes, offset : Nat, output : Bytes], result: Bytes,
 body: (match fuel | Nat.zero => output | Nat.succ remaining =>

definition{
 name: normalize, parameters: [value : String], result: String,

definition{
 name: unigrams, parameters: [tokens : (List String)], result: MatterText::TextItems,

definition{
 name: pairs, parameters: [tokens : (List String)], result: MatterText::TextItems,

definition{
 name: appendFeatures, parameters: [left : (List String), right : (List String)], result: MatterText::TextItems,

definition{
 name: nonemptyTokens, parameters: [tokens : (List String)], result: MatterText::TextItems,
 body: (match tokens | List.nil => (record MatterText::TextItems {values := (nil String)}) | List.cons token rest =>
 (let following : MatterText::TextItems := (nonemptyTokens rest); (if (prim equal token "" : Bool) then following else (record MatterText::TextItems {values := (cons token (project following values))})))),

definition{
 name: features, parameters: [text : String], result: MatterText::TextItems,
 body: (match (prim split_exact (normalize text) " " (256 : UInt32) : (Option (List String)))
 | Option.none => (record MatterText::TextItems {values := (nil String)})
 | Option.some raw => (let tokens : MatterText::TextItems := (nonemptyTokens raw);
 (MatterText::distinctWords (project (appendFeatures (project (unigrams (project tokens values)) values) (project (pairs (project tokens values)) values)) values)))),
 executable: true, axioms: [Classical.choice, Quot.sound, propext]
}

definition{
 name: strengthen, parameters: [synapses : (List Synapse), feature : String], result: Synapses,
 body: (match synapses | List.nil => (record Synapses {values := (cons (record Synapse {feature := feature, weight := 1}) (nil Synapse))})
 | List.cons synapse rest => (if (prim equal (project synapse feature) feature : Bool) then
 (record Synapses {values := (cons (record Synapse {feature := feature, weight := ((project synapse weight) + 1)}) rest)})
 else (let following : Synapses := (strengthen rest feature); (record Synapses {values := (cons synapse (project following values))})))),
 executable: true, axioms: [], recursive_argument: synapses
}

definition{
 name: learnFeatures, parameters: [activeFeatures : (List String), synapses : Synapses], result: Synapses,

 executable: true, axioms: [], recursive_argument: activeFeatures
}

definition{
 name: learnExample, parameters: [neurons : (List Neuron), node : Matter::Node, activeFeatures : (List String)], result: NeuralModel,
 body: (match neurons
 | List.nil => (record NeuralModel {neurons := (cons (record Neuron {facet := (project node facet), observations := 1, synapses := (project (learnFeatures activeFeatures (record Synapses {values := (nil Synapse)})) values), dependencies := (cons (Matter::nodeReference node) (nil Matter::Reference))}) (nil Neuron))})
 | List.cons neuron rest => (if (prim equal (project neuron facet) (project node facet) : Bool) then
 (record NeuralModel {neurons := (cons (record Neuron {facet := (project neuron facet), observations := ((project neuron observations) + 1), synapses := (project (learnFeatures activeFeatures (record Synapses {values := (project neuron synapses)})) values), dependencies := (cons (Matter::nodeReference node) (project neuron dependencies))}) rest)})
 else (let following : NeuralModel := (learnExample rest node activeFeatures); (record NeuralModel {neurons := (cons neuron (project following neurons))})))),
 executable: true, axioms: [], recursive_argument: neurons
}

definition{
 name: trainLoop, parameters: [nodes : (List Matter::Node), admitted : (List Matter::Reference), network : NeuralModel], result: NeuralModel,
 body: (match nodes | List.nil => network | List.cons node rest =>
 (if ((Matter::containsReference admitted (Matter::nodeReference node)) && (not (prim equal (project node facet) "rule" : Bool))) then
 (trainLoop rest admitted (learnExample (project network neurons) node (project (features (project node value)) values))) else (trainLoop rest admitted network))),
 executable: true, axioms: [Classical.choice, Quot.sound, propext], recursive_argument: nodes
}

definition{
 name: train, parameters: [nodes : (List Matter::Node), sources : (List Matter::Source)], result: NeuralModel,

 executable: true, axioms: [Classical.choice, Quot.sound, propext]
}

definition{
 name: excite, parameters: [synapses : (List Synapse), query : (List String)], result: Nat,
 body: (match synapses | List.nil => 0 | List.cons synapse rest =>
 ((if (MatterText::wordPresent query (project synapse feature)) then (project synapse weight) else 0) + (excite rest query))),
 executable: true, axioms: [], recursive_argument: synapses
}

definition{
 name: activate, parameters: [neurons : (List Neuron), query : (List String), best : Activation], result: Activation,
 body: (match neurons | List.nil => best | List.cons neuron rest =>
 (let strength : Nat := (excite (project neuron synapses) query);
 (let proposed : Nat := (prim multiply strength (project best observations) : Nat);
 (let previous : Nat := (prim multiply (project best excitation) (project neuron observations) : Nat);
 (activate rest query
 (if (previous <? proposed) then (record Activation {facet := (project neuron facet), excitation := strength, observations := (project neuron observations), ambiguous := false, dependencies := (project neuron dependencies)}) else
 (if ((0 <? strength) && ((proposed =? previous) && (not (prim equal (project neuron facet) (project best facet) : Bool)))) then
 (record Activation {facet := (project best facet), excitation := (project best excitation), observations := (project best observations), ambiguous := true, dependencies := (project best dependencies)}) else best))))))),
 executable: true, axioms: [], recursive_argument: neurons
}

definition{
 name: infer, parameters: [network : NeuralModel, text : String], result: Activation,

 executable: true, axioms: [Classical.choice, Quot.sound, propext]
}

}

\begin{lexlean}{MatterResources}
\useglossary{lexlean.std.bool@1.2.0}
\useglossary{lexlean.std.nat@1.2.0}
\useglossary{prism.arch@1.0.0}
\importmodule{Matter}
\importmodule{MatterText}
\importmodule{MatterCsv}
\importmodule{Foundation.Bytes}
\importmodule{Foundation.Utf8}
\title{Text application model}
\begin{semanticmodule}
structure{
 name: SourceItem, type_parameters: [], parameters: [],
 fields: [{name: location, type: String}, {name: heading, type: String}, {name: value, type: String}, {name: start, type: Nat}]
}
structure{
 name: SourceItems, type_parameters: [], parameters: [], fields: [{name: values, type: (List SourceItem)}]
}
structure{
 name: SourceContent, type_parameters: [], parameters: [],
 fields: [{name: items, type: (List SourceItem)}, {name: scope, type: String}, {name: focusIndex, type: Nat}]
}
definition{
 name: sourceItemCount, parameters: [items : (List SourceItem)], result: Nat,
 body: (match items | List.nil => 0 | List.cons item rest => (1 + (sourceItemCount rest))),
 executable: true, axioms: [], recursive_argument: items
}
definition{
 name: sourceItemIndex, parameters: [items : (List SourceItem), start : Nat, index : Nat], result: Nat,
 body: (match items | List.nil => 0 | List.cons item rest => (if ((project item start) =? start) then index else (sourceItemIndex rest start (index + 1)))),
 executable: true, axioms: [], recursive_argument: items
}
definition{
 name: csvRowAt, parameters: [cells : (List MatterCsv::CsvCell), start : Nat], result: Nat,
 body: (match cells | List.nil => 0 | List.cons cell rest => (if ((project cell start) =? start) then (project cell row) else (csvRowAt rest start))),
 executable: true, axioms: [], recursive_argument: cells
}
definition{
 name: prependCsvSourceItem, parameters: [cell : MatterCsv::CsvCell, headers : (List String), row : Nat, following : SourceItems], result: SourceItems,
 body: (if (((project cell row) =? 1) || ((not (row =? 0)) && (not ((project cell row) =? row)))) then following else
 (record SourceItems {values := (cons (record SourceItem {
 location := (MatterText::concat "row " (MatterText::concat (MatterText::decimalLoop 20 (project cell row)) (MatterText::concat " · column " (MatterText::decimalLoop 20 (project cell column))))),
 heading := (MatterCsv::csvFacetAt headers (prim subtract (project cell column) 1 : Nat)), value := (project cell value), start := (project cell start)}) (project following values))})),
 executable: true, axioms: [propext]
}
definition{
 name: csvSourceItems, parameters: [cells : (List MatterCsv::CsvCell), headers : (List String), row : Nat], result: SourceItems,
 body: (match cells | List.nil => (record SourceItems {values := (nil SourceItem)}) | List.cons cell rest =>
 (prependCsvSourceItem cell headers row (csvSourceItems rest headers row))),
 executable: true, axioms: [propext], recursive_argument: cells
}
definition{
 name: prependTextSourceItem, parameters: [line : String, offset : Nat, ordinal : Nat, following : SourceItems], result: SourceItems,
 body: (record SourceItems {values := (cons (record SourceItem {location := (MatterText::concat "line " (MatterText::decimalLoop 20 ordinal)), heading := "", value := line, start := offset}) (project following values))}),
 executable: true, axioms: [propext]
}
definition{
 name: textSourceItems, parameters: [lines : (List String), offset : Nat, ordinal : Nat], result: SourceItems,
 body: (match lines | List.nil => (record SourceItems {values := (nil SourceItem)}) | List.cons line rest =>
 (prependTextSourceItem line offset ordinal (textSourceItems rest (offset + ((Foundation.Bytes::byteLength (Foundation.Utf8::encode line)) + 1)) (ordinal + 1)))),
 executable: true, axioms: [propext], recursive_argument: lines
}
definition{
 name: resourceContent, parameters: [source : Matter::Source, csv : Bool, focus : (Option Matter::Citation)], result: (Option SourceContent),
 body: (if csv then (match (MatterCsv::parseCsv (project source content)) | Option.none => (ctor Option.none [SourceContent]) | Option.some table =>
 (let row : Nat := (match focus | Option.none => 0 | Option.some citation => (csvRowAt (project table values) (project citation start)));
 (let items : SourceItems := (csvSourceItems (project table values) (project (MatterCsv::csvHeaders (project table values)) values) row);
 (ctor Option.some [SourceContent] (record SourceContent {items := (project items values), scope := (if (row =? 0) then "CSV source" else (MatterText::concat "CSV source · row " (MatterText::decimalLoop 20 row))), focusIndex := (match focus | Option.none => 0 | Option.some citation => (sourceItemIndex (project items values) (project citation start) 0))}))))) else
 (match (Foundation.Utf8::decode (project source content)) | Option.none => (ctor Option.none [SourceContent]) | Option.some text =>
 (match (prim split_exact text "\n" (64 : UInt32) : (Option (List String))) | Option.none => (ctor Option.none [SourceContent]) | Option.some lines =>
 (let items : SourceItems := (textSourceItems lines 0 1);
 (ctor Option.some [SourceContent] (record SourceContent {items := (project items values), scope := "Text source", focusIndex := (match focus | Option.none => 0 | Option.some citation => (sourceItemIndex (project items values) (project citation start) 0))})))))),
 executable: true, axioms: [Classical.choice, Quot.sound, propext]
}
\end{semanticmodule}
\end{lexlean}

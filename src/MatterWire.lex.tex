definition{
  name: escapeLoop, parameters: [fuel : Nat, value : Bytes, offset : Nat, escaped : Bytes], result: Bytes,
  body: (match fuel
    | Nat.zero => escaped
    | Nat.succ remaining => (match (Foundation.Bytes::sliceBytes value offset 1)
      | Option.none => escaped
      | Option.some byte => (escapeLoop remaining value (offset + 1) (prim append escaped (escapeByte byte) : Bytes)))),
  recursive_argument: fuel,
  executable: true
}

definition{
  name: quote, parameters: [value : String], result: String,
  body: (let bytes : Bytes := (Foundation.Utf8::encode value);
    (match (Foundation.Utf8::decode (escapeLoop (Foundation.Bytes::byteLength bytes) bytes 0 0x))
      | Option.none => "\"\""
      | Option.some escaped => (MatterText::concat "\"" (MatterText::concat escaped "\"")))),
  axioms: [Classical.choice, Quot.sound, propext],
  executable: true
}

definition{
  name: quoteItems, parameters: [values : (List String)], result: MatterText::TextItems,
